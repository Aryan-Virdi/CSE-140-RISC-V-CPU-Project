#ifndef WRITEBACK_HPP
#define WRITEBACK_HPP
#include "printing.hpp"

void writeback(int alu_result, int mem_read_data, bool reg_write, bool mem_to_reg, int register_file[32], int destination_register, int& total_clock_cycles, PrintEvent& printQueue, bool is_jump, int pc_plus_4){
    if (reg_write) {
        int value; 

        if (is_jump) {
            value = pc_plus_4; // For JAL, write PC+4 to the register
        } else (mem_to_reg){
            value = mem_read_data // LW
        }
        else {
            value = alu_result; // R-type or I-type arithmetic
        }
        register_file[destination_register] = value;

        printQueue.addPrintEvent(LocationType::registerFile, destination_register, value);
    }

    total_clock_cycles++; // Increment clock cycles for writeback stage
}

#endif

// Writeback stage: selects data to write into register
// mem_to_reg = 1 -> memory, 0 -> ALU
